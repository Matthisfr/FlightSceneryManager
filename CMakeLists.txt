cmake_minimum_required(VERSION 3.21)
project(FlightSceneryManager VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(FlightSceneryManager WIN32 src/main.cpp)
target_link_libraries(FlightSceneryManager PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
 target_compile_options(FlightSceneryManager PRIVATE /W4 /utf-8)
endif()
